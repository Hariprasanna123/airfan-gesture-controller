function isClosedFist(landmarks) { // Reject hands that are too small or too far from the camera. const palmWidth = distance(landmarks[5], landmarks[17]); const palmLength = distance(landmarks[0], landmarks[9]); if (palmWidth < 0.09 || palmLength < 0.05) { return false; } // Approximate the centre of the palm. const palmPoints = [ landmarks[0], landmarks[5], landmarks[9], landmarks[13], landmarks[17] ]; const palmCenter = palmPoints.reduce( (sum, point) => ({ x: sum.x + point.x / palmPoints.length, y: sum.y + point.y / palmPoints.length, z: sum.z + (point.z || 0) / palmPoints.length }), { x: 0, y: 0, z: 0 } ); // Index, middle, ring and little finger joint landmarks. const fingers = [ { mcp: 5, pip: 6, dip: 7, tip: 8 }, { mcp: 9, pip: 10, dip: 11, tip: 12 }, { mcp: 13, pip: 14, dip: 15, tip: 16 }, { mcp: 17, pip: 18, dip: 19, tip: 20 } ]; let foldedCount = 0; for (const finger of fingers) { const mcp = landmarks[finger.mcp]; const pip = landmarks[finger.pip]; const dip = landmarks[finger.dip]; const tip = landmarks[finger.tip]; // A bent PIP or DIP joint is evidence that the finger is folded. const pipAngle = angleAtJoint(mcp, pip, dip); const dipAngle = angleAtJoint(pip, dip, tip); // Check whether the fingertip has curled back towards the palm. const tipToPalm = distance(tip, palmCenter) / palmWidth; const tipToMcp = distance(tip, mcp) / palmWidth; const jointsFolded = pipAngle < 145 || dipAngle < 140; const tipNearPalm = tipToPalm < 0.95 || tipToMcp < 0.75; if (jointsFolded && tipNearPalm) { foldedCount++; } } // Require all four main fingers to be folded. if (foldedCount !== 4) { return false; } // The thumb should be tucked near the palm or across the fingers. const thumbTip = landmarks[4]; const thumbIp = landmarks[3]; const thumbMcp = landmarks[2]; const thumbTipToPalm = distance(thumbTip, palmCenter) / palmWidth; const thumbTipToIndexMcp = distance(thumbTip, landmarks[5]) / palmWidth; const thumbAngle = angleAtJoint(thumbMcp, thumbIp, thumbTip); const thumbTucked = thumbTipToPalm < 1.15 || thumbTipToIndexMcp < 0.85 || thumbAngle < 135; return thumbTucked; } function getStrictFingerCount(landmarks) { const palmSize = distance(landmarks[0], landmarks[9]); if (palmSize < 0.09) { return null; } // Check for a closed fist first, before normal finger counting. if (isClosedFist(landmarks)) { return 0; // OFF } const thumb = classifyThumb(landmarks); const index = classifyFinger(landmarks, 5, 6, 7, 8); const middle = classifyFinger(landmarks, 9, 10, 11, 12); const ring = classifyFinger(landmarks, 13, 14, 15, 16); const pinky = classifyFinger(landmarks, 17, 18, 19, 20); const states = [thumb, index, middle, ring, pinky]; // Unclear finger positions must never produce a speed change. if (states.some(state => state === null)) { return null; } return states.filter(Boolean).length; }
